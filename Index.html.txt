<script>
    // 1. LOGIN LOGIKA
    function login() {
        const email = document.getElementById('login-email').value;
        if(email.includes('@')) {
            document.getElementById('auth-screen').style.display = 'none';
            document.getElementById('app-screen').style.display = 'block';
            document.getElementById('display-email').innerText = "Aktivní skaut: " + email;
        } else { alert("Zadejte platný pracovní email."); }
    }

    // 2. CHART LOGIKA (Radar Chart)
    const ctx = document.getElementById('scoutChart').getContext('2d');
    let myChart = new Chart(ctx, {
        type: 'radar',
        data: {
            labels: ['Rychlost', 'Technika', 'Střelba', 'Fyzika', 'Taktika'],
            datasets: [{
                label: 'Hodnocený hráč',
                data: [70, 60, 50, 65, 55],
                backgroundColor: 'rgba(26, 115, 232, 0.2)',
                borderColor: '#1a73e8',
                borderWidth: 3,
                pointBackgroundColor: '#1a73e8'
            }, {
                label: 'Benchmark (Průměr pozice)',
                data: [75, 75, 70, 70, 70],
                backgroundColor: 'rgba(0, 0, 0, 0.05)',
                borderColor: 'rgba(0, 0, 0, 0.2)',
                borderWidth: 1,
                borderDash: [5, 5]
            }]
        },
        options: { 
            scales: { r: { min: 0, max: 100, ticks: { display: false } } },
            plugins: { legend: { position: 'bottom' } }
        }
    });

    function refresh() {
        const vals = [
            parseInt(document.getElementById('v1').value),
            parseInt(document.getElementById('v2').value),
            parseInt(document.getElementById('v3').value),
            parseInt(document.getElementById('v4').value),
            parseInt(document.getElementById('v5').value)
        ];
        myChart.data.datasets[0].data = vals;
        myChart.update();

        // Výpočet průměru
        const score = Math.round(vals.reduce((a, b) => a + b) / vals.length);
        document.getElementById('avg-score').innerText = score;
    }

    function syncName() {
        const name = document.getElementById('p-name').value;
        document.getElementById('display-name').innerText = name || "Nový report";
    }

    // 3. FOTO LOGIKA
    function loadPhoto(input) {
        if (input.files && input.files[0]) {
            const reader = new FileReader();
            reader.onload = e => document.getElementById('preview').src = e.target.result;
            reader.readAsDataURL(input.files[0]);
        }
    }

    // 4. PDF EXPORT
    async function generatePDF() {
        const { jsPDF } = window.jspdf;
        const report = document.getElementById('report-to-pdf');
        
        // Vytvoření snímku obrazovky
        const canvas = await html2canvas(report, { scale: 2 });
        const imgData = canvas.toDataURL('image/png');
        
        const pdf = new jsPDF('p', 'mm', 'a4');
        const width = pdf.internal.pageSize.getWidth();
        const height = (canvas.height * width) / canvas.width;
        
        pdf.text("OFFICIAL SCOUTING REPORT", 105, 10, { align: "center" });
        pdf.addImage(imgData, 'PNG', 0, 20, width, height);
        
        const name = document.getElementById('p-name').value || "Hrac";
        pdf.save(`Report_${name}.pdf`);
    }
</script>
